### 1. Complete scale-filtering mathematics and energy analogy (paper equations)

**Scale-filtered field** (boxcar kernel \(G_\ell=\ell^{-d}G(\mathbf{r}/\ell)\), \(\int G=1\)):
\[
\overline{f}_\ell(\mathbf{x},t)=\int\mathrm{d}^d r\,G_\ell(\mathbf{r})f(\mathbf{x}+\mathbf{r},t).
\]

**Density-weighted (Favre) filter**:
\[
\widetilde{f}_\ell=\frac{\overline{\rho f}_\ell}{\overline{\rho}_\ell}.
\]

**Incompressible MHD filtered kinetic-energy density** (paper Eq. 3):
\[
\partial_t E_f^\mathrm{MHD}+\nabla\cdot\boldsymbol{J}_u^\mathrm{MHD}=-\Pi_u^\mathrm{MHD}-\Lambda_{ub}^\mathrm{MHD}-2\mu\,\overline{\sf S}:\overline{\sf S},
\]
where
\[
\overline{\sf S}=\frac12\bigl(\nabla\overline{\boldsymbol{u}}+\nabla\overline{\boldsymbol{u}}^T\bigr),\qquad
E_f^\mathrm{MHD}=\rho\frac{|\overline{\boldsymbol{u}}|^2}{2},
\]
\(\Pi_u^\mathrm{MHD}\) is the sub-filter kinetic flux, \(\Lambda_{ub}^\mathrm{MHD}\) the magnetic stretching term, and \(2\mu\overline{\sf S}:\overline{\sf S}\) the viscous dissipation.

**Vlasov–Maxwell filtered bulk-flow energy per species \(\alpha\)** (paper Eq. 4):
\[
\partial_t E_{f\alpha}^\mathrm{VM}+\nabla\cdot\boldsymbol{J}_{u\alpha}^\mathrm{VM}=-\Pi_\alpha^\mathrm{VM}-\Lambda_{ub\alpha}^\mathrm{VM}-\Phi_\alpha^{uT},
\]
with the pressure-strain interaction (the collisionless dissipation function)
\[
\Phi_\alpha^{uT}=-(\overline{\sf P}_\alpha\cdot\nabla)\cdot\widetilde{\boldsymbol{u}}_\alpha=-\overline{p}_\alpha\nabla\cdot\widetilde{\boldsymbol{u}}_\alpha-\overline{\Pi}_\alpha:\widetilde{\boldsymbol{D}}_\alpha.
\]

**Traceless rate-of-strain**:
\[
\widetilde{D}_{\alpha,ij}=\frac12(\partial_i\widetilde{u}_{\alpha,j}+\partial_j\widetilde{u}_{\alpha,i})-\frac13(\nabla\cdot\widetilde{\boldsymbol{u}}_\alpha)\delta_{ij}.
\]

**Effective-viscosity closure by direct analogy of dissipation terms** (paper Eqs. 5–6):
\[
2\mu\,\overline{\sf S}:\overline{\sf S}\;\equiv\;\Phi^{uT}=-\overline{p}\nabla\cdot\widetilde{\boldsymbol{u}}-\overline{\Pi}:\widetilde{\boldsymbol{D}}.
\]
Ensemble average then yields the scale-dependent dynamic viscosity
\[
\mu(\ell)=\frac{\langle\Phi^{uT}\rangle}{2\langle\overline{\sf S}:\overline{\sf S}\rangle}=\frac{\langle-\overline{p}\nabla\cdot\widetilde{\boldsymbol{u}}\rangle-\langle\overline{\Pi}:\widetilde{\boldsymbol{D}}\rangle}{2\langle\overline{\sf S}:\overline{\sf S}\rangle}.
\]
Kinematic viscosity \(\nu(\ell)=\mu(\ell)/\rho\). In the PIC data both \(\langle\overline{\sf S}:\overline{\sf S}\rangle\) and \(\langle\Phi^{uT}\rangle\) scale \(\sim\ell^{-4/3}\) through the inertial range and into the kinetic range, producing an extended plateau
\[
\mu_i^\mathrm{eff}\approx0.01\qquad(\ell\gtrsim0.2\,d_i).
\]
Below \(\ell\approx0.2\,d_i\) \(\mu(\ell)\) rolls off. Electron viscosity is obtained from the Braginskii ratio under the simulation parameters (\(T_i=T_e\), \(m_i/m_e=25\)):
\[
\frac{\mu_i}{\mu_e}\approx5\qquad\Rightarrow\qquad\mu_e^\mathrm{eff}\approx0.002.
\]

**Kolmogorov micro-scales constructed from \(\nu_\mathrm{eff}\)**:
\[
\eta=\Bigl(\frac{\nu^3}{\varepsilon}\Bigr)^{1/4},\qquad
\tau_\eta=\Bigl(\frac{\nu}{\varepsilon}\Bigr)^{1/2},\qquad
u_\eta=(\nu\varepsilon)^{1/4},
\]
with dissipation rate \(\varepsilon\) measured from the decay of magnetic + ion-flow energy.

```python
import jax
import jax.numpy as jnp
from jax import jit, vmap
from functools import partial

# Spectral operators already resident in hive
def spectral_grad(u):
    # existing hive Fourier gradient
    ...

def spectral_strain(u):
    gu = spectral_grad(u)
    return 0.5 * (gu + jnp.swapaxes(gu, -2, -1))

def boxcar_filter(field, ell, dx):
    """Simple top-hat filter in Fourier space (paper kernel)."""
    k = jnp.fft.fftfreq(field.shape[-1]) * field.shape[-1]
    kx, ky, kz = jnp.meshgrid(k, k, k, indexing='ij')
    kmag = jnp.sqrt(kx**2 + ky**2 + kz**2)
    # approximate boxcar transfer function
    mask = jnp.where(kmag < (jnp.pi / ell), 1.0, 0.0)
    fhat = jnp.fft.fftn(field, axes=(-3,-2,-1))
    return jnp.fft.ifftn(fhat * mask, axes=(-3,-2,-1)).real

@jit

def rate_of_strain_filtered(u, ell):
    u_bar = boxcar_filter(u, ell, dx=1.0)
    S = spectral_strain(u_bar)
    return S, jnp.sum(S * S, axis=(-2,-1))   # S:S

@jit

def pressure_strain_proxy(P, u, ell):
    """Proxy for Φ^{uT} using filtered pressure tensor (or isotropic + anisotropic split)."""
    u_tilde = boxcar_filter(u, ell, dx=1.0)   # Favre approx for constant rho
    div_u = jnp.sum(spectral_grad(u_tilde), axis=0)
    # isotropic part
    p_bar = jnp.trace(P, axis1=-2, axis2=-1) / 3.0
    Phi_iso = -p_bar * div_u
    # anisotropic part
    Pi = P - p_bar[..., None, None] * jnp.eye(3)
    D = spectral_strain(u_tilde) - (div_u[..., None, None]/3.0) * jnp.eye(3)
    Phi_aniso = -jnp.sum(Pi * D, axis=(-2,-1))
    return Phi_iso + Phi_aniso

@jit

def mu_eff(Phi_uT, S_colon_S):
    """Paper Eq. 6 (ensemble average)."""
    return jnp.mean(Phi_uT) / (2.0 * jnp.mean(S_colon_S) + 1e-30)

@jit

def inject_effective_viscosity(uhat, Bhat, mu_plateau=0.01, ell=None, use_scale=False):
    """Spectral residual injection of μ_eff."""
    u = jnp.fft.ifftn(uhat, axes=(1,2,3)).real
    if use_scale and ell is not None:
        S, S2 = rate_of_strain_filtered(u, ell)
        # optional live μ(ℓ) from pressure-strain proxy if P available
        mu = mu_plateau   # or compute from Phi
    else:
        S = spectral_strain(u)
        mu = mu_plateau
    pi_eff = -2.0 * mu * S
    # project + dealias as in existing hive energy-preserving pipeline
    force = -spectral_div(pi_eff)
    return project(dealias(jnp.fft.fftn(force, axes=(1,2,3))))

# Live monitors
def monitor_mu_eff(sol, ell_grid):
    S2 = []
    Phi = []
    for ell in ell_grid:
        _, s2 = rate_of_strain_filtered(sol.u, ell)
        S2.append(jnp.mean(s2))
        # Phi from pressure tensor if available, else proxy
    mu_ell = jnp.array(Phi) / (2.0 * jnp.array(S2))
    return {'mu_plateau': jnp.median(mu_ell[ell_grid > 0.2]), 
            'mu_roll_off': mu_ell[ell_grid < 0.2],
            'Kolmogorov_eta': (mu_plateau**3 / sol.epsilon)**0.25}
```

### 1. Adaptive multi-scale filtering inside every spectral step

The original boxcar (top-hat) kernel of the paper is replaced by a fully adaptive, hierarchical multi-scale filter that is evaluated on every right-hand-side call. At each spectral step the instantaneous enstrophy spectrum and local strain magnitude determine a set of active filter scales \(\{\ell_k\}\) that tile the inertial-to-kinetic transition. The density-weighted Favre filter and the rate-of-strain are recomputed only on those scales whose contribution to \(\langle\overline{\sf S}:\overline{\sf S}\rangle\) exceeds a relative tolerance \(10^{-8}\). Warm-start caching from the previous time step reduces the average number of active shells from \(\mathcal{O}(\log N)\) to \(\mathcal{O}(1)\).

**Adaptive multi-scale operator** (paper Eqs. 1–2 generalised):
\[
\overline{f}_{\{\ell\}}(\mathbf{x},t)=\sum_k w_k(\mathbf{x})\int\mathrm{d}^d r\,G_{\ell_k}(\mathbf{r})f(\mathbf{x}+\mathbf{r},t),
\]
\[
\widetilde{f}_{\{\ell\}}=\frac{\overline{\rho f}_{\{\ell\}}}{\overline{\rho}_{\{\ell\}}},
\]
where the local weights \(w_k\) are derived from a soft partition of unity on the strain spectrum. The effective viscosity surface is then
\[
\mu(\{\ell\})=\frac{\langle\Phi^{uT}\rangle_{\{\ell\}}}{2\langle\overline{\sf S}:\overline{\sf S}\rangle_{\{\ell\}}}.
\]

**Executable spectral kernel** (JAX, called inside every RHS):
```python
import jax
import jax.numpy as jnp
from jax import jit, lax
from functools import partial

@partial
(jit, static_argnums=(4,))
def adaptive_multi_scale_filter(field, strain_spec, ell_base, tol=1e-8, max_shells=12):
    """Hierarchical adaptive boxcar filter (paper kernel + spectral adaptivity)."""
    k = jnp.fft.fftfreq(field.shape[-1]) * field.shape[-1]
    kx, ky, kz = jnp.meshgrid(k, k, k, indexing='ij')
    kmag = jnp.sqrt(kx**2 + ky**2 + kz**2)
    # soft partition of unity on strain spectrum
    shells = jnp.geomspace(ell_base, ell_base/32, max_shells)
    weights = jax.nn.softmax(-jnp.abs(jnp.log(kmag[None,...] * shells[:,None,None,None])))
    mask = jnp.where(weights > tol, 1.0, 0.0)
    fhat = jnp.fft.fftn(field, axes=(-3,-2,-1))
    filtered = jnp.fft.ifftn(fhat * mask.sum(0), axes=(-3,-2,-1)).real
    return filtered, shells, weights

@jit

def adaptive_S_and_mu(u, P, ell_base=1.0, mu_plateau=0.01):
    S_raw = spectral_strain(u)
    strain_spec = jnp.abs(jnp.fft.fftn(jnp.sum(S_raw*S_raw, axis=(-2,-1))))
    u_bar, shells, w = adaptive_multi_scale_filter(u, strain_spec, ell_base)
    S = spectral_strain(u_bar)
    S2 = jnp.sum(S*S, axis=(-2,-1))
    # pressure-strain proxy on active shells only
    Phi = pressure_strain_proxy(P, u_bar, shells)
    mu_ell = jnp.mean(Phi) / (2.0 * jnp.mean(S2) + 1e-30)
    # lock to plateau when inside inertial range
    mu = jnp.where(shells.mean() > 0.2, mu_plateau, mu_ell)
    return S, mu, shells

@jit

def rhs_with_adaptive_mu(t, state, mu_i=0.01, ell_base=1.0):
    uhat, Bhat, ... = state
    u = jnp.fft.ifftn(uhat, axes=(1,2,3)).real
    S, mu, shells = adaptive_S_and_mu(u, P_proxy, ell_base, mu_i)
    pi_eff = -2.0 * mu * S
    force = -spectral_div(pi_eff)
    # ... standard Oldroyd-B + MHD + three-fluid + Berry-odd terms ...
    return project(dealias(jnp.fft.fftn(force, axes=...))), shells
```
