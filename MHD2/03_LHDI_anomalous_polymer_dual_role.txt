def energy_preserving_filter(u, k_cutoff, dx):
    # u: (3,Nx,Ny,Nz)
    Ek = 0.5 * np.sum(u**2) * dx**3
    u_hat = np.fft.fftn(u, axes=(1,2,3))
    kx,ky,kz = ...  # wavenumber grids
    kmag = np.sqrt(kx**2+ky**2+kz**2)
    mask = (kmag <= k_cutoff)
    u_star = np.fft.ifftn(u_hat * mask, axes=(1,2,3)).real
    Ek_star = 0.5 * np.sum(u_star**2) * dx**3
    scale = np.sqrt(Ek / (Ek_star + 1e-30))
    u_safe = u_star * scale
    # optional: re-project div-free (Fourier exact)
    return u_safe
```

Call once per stage **before** induction / Lorentz / polymer advection; never after. Hyperviscosity (order \(2p\)) may be applied only to the highest 5–10 % of modes on velocity if odd-viscous waves pile up.

def sheet_thickness(J, B, dBdx, dBdy, dBdz):
    Jmag = np.sqrt(np.sum(J**2, axis=0))
    idx = np.unravel_index(np.argmax(Jmag), Jmag.shape)
    gradB = np.stack([dBdx[...,idx], dBdy[...,idx], dBdz[...,idx]], axis=1)  # 3x3
    svals = np.linalg.svd(gradB, compute_uv=False)
    lambda_min = 1.0 / (svals[0] + 1e-30)
    return lambda_min, idx, Jmag[idx]

def gradient_relaxation_proxy(J, L_box):
    # width of |J| profile or variance of current-sheet edges
    Jy = J[1]
    profile = np.mean(np.abs(Jy), axis=(0,1))  # average along tube
    # FWHM or second-moment width
    return second_moment_width(profile)

def kink_amplitude(B, k_low_cutoff):
    # low-k power in transverse B or centre-line displacement of max|J|
    By_hat = np.fft.fftn(B[1])
    return np.sum(np.abs(By_hat[kmag < k_low_cutoff])**2)

# Fourier operators (periodic, 2/3 dealias)
N = 48  # or 64/96 for convergence
k = np.fft.fftfreq(N)*N
kx,ky,kz = np.meshgrid(k,k,k,indexing='ij')
k2 = kx**2+ky**2+kz**2; k2[0,0,0]=1.0
ik = 1j*np.stack([kx,ky,kz])

def project(uhat):
    kdu = np.sum(ik*uhat, axis=0)
    return uhat - ik*(kdu/k2)

def rhs_LHDI_polymer(t, state, eta_odd, alpha_LHDI, beta_LHDI, C_LHDI,
                     eta_s=1e-3, eta_p=0.1, lambda_p=1.0, eta_m=1e-3):
    uhat, Bhat, psihat = state  # psi = log-conformation
    u = np.fft.ifftn(uhat, axes=(1,2,3)).real
    B = np.fft.ifftn(Bhat, axes=(1,2,3)).real
    psi = np.fft.ifftn(psihat, axes=(1,2,3)).real
    C = matrix_exp(psi)          # positive-definite
    T = (eta_p/lambda_p)*(C - np.eye(3)[None,None,None,...])
    # odd/anisotropic augmentation
    omega = curl(u)
    b = B / (np.linalg.norm(B,axis=0)+1e-30)
    T_odd = eta_odd * antisym_outer(omega, u)
    T_aniso = alpha_LHDI * outer(b, b·grad(u)) + beta_LHDI * sym_perp(J,B)
    T_tot = T + T_odd + T_aniso
    # momentum
    adv = u·grad(u)
    Lorentz = cross(J, B)
    divT = div(T_tot)
    du = -adv -grad(p) + eta_s*lap(u) + divT + Lorentz
    duhat = project(np.fft.fftn(du, axes=(1,2,3)))
    # induction — RAW or energy-preserving u only
    u_ind = energy_preserving_filter(u, k_cutoff=0.8*kmax, dx=dx)
    dB = curl(cross(u_ind, B)) + eta_tot * lap(B)   # eta_tot includes eta_LHDI(T)
    dBhat = dealias(np.fft.fftn(dB, axes=(1,2,3)))
    # conformation (Upper-convected + anisotropic source)
    dpsi = upper_convected(psi, u, T) + anisotropic_source(alpha_LHDI,...)
    dpsihat = dealias(np.fft.fftn(dpsi, axes=(1,2,3)))
    return duhat, dBhat, dpsihat
```
