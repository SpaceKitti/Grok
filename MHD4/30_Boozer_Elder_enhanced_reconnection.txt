@torch.jit.script
def stream_function(x: torch.Tensor, y: torch.Tensor, t: torch.Tensor,
                    a: float, tau: float, lam2: float = 0.0) -> torch.Tensor:
    r2 = (x * x + y * y) / (a * a)
    envelope = (1.0 - r2).clamp(min=0.0).pow(3) * torch.exp(-lam2 * r2)
    h_tilde = (a * a / tau) * (
        0.25 * torch.cos(6.0 * torch.pi * t / tau) * torch.cos(2.0 * torch.pi * x / a)
        + 0.25 * torch.cos(4.0 * torch.pi * t / tau) * torch.cos(2.0 * torch.pi * y / a)
    )
    return h_tilde * envelope

@torch.jit.script
def footpoint_velocity(h: torch.Tensor, dx: float) -> tuple:
    # v = z-hat × ∇h  →  (∂h/∂y, -∂h/∂x)
    dhdx = torch.gradient(h, spacing=dx, dim=-1)[0]
    dhdy = torch.gradient(h, spacing=dx, dim=-2)[0]
    return dhdy, -dhdx

@torch.jit.script
def jacobian_frobenius(J: torch.Tensor) -> torch.Tensor:
    # J: (..., 2, 2)
    return torch.sqrt((J * J).sum(dim=(-2, -1)))

@torch.jit.script
def lyapunov_ratio(J: torch.Tensor) -> torch.Tensor:
    # SVD of 2×2 Jacobian; Λ_u / Λ_s = Δ_max / Δ_min
    s = torch.linalg.svdvals(J)
    return (s[..., 0] / (s[..., 1] + 1e-30)).clamp(min=1.0)

@torch.jit.script
def parallel_current_K(Omega: torch.Tensor, t: torch.Tensor, L: float) -> torch.Tensor:
    return Omega * t / L

@torch.jit.script
def boozer_elder_residual(
    J: torch.Tensor, Omega: torch.Tensor, t: torch.Tensor,
    L: float, Rm: float, tau_ev: float
) -> dict:

    sigma_num = 0.5 * torch.log(ratio)
    R_Lambda = (sigma_num - t / tau_ev).abs().mean()
    K = parallel_current_K(Omega, t, L)
    R_K = (K - Omega.max() * t / L).abs().mean()
    R_rec = torch.relu(torch.log(torch.tensor(Rm)) - torch.log(ratio.max()))
    return {
        "R_Lambda": R_Lambda,
        "R_K": R_K,
        "R_rec": R_rec,
        "Delta_max_over_min": ratio.max(),
        "K_max": K.abs().max(),
        "scar_floor_feed": R_Lambda + R_K + R_rec,
        "holographic_DeltaF_trigger": torch.relu(R_rec),
        "Qin_compatible": J
    }
