### Production kernels (Torch / JAX, Qin-drop-in + spectral)

```python
import torch

@torch.jit.script
def bloch_H(kx: torch.Tensor, ky: torch.Tensor,
            t1: float = -1.0, t2: float = -0.5,
            ty: float = -1.0, gamma: float = 0.2) -> torch.Tensor:
    dx = (t1 + t2) * torch.cos(kx) + 2.0 * ty * torch.cos(ky)
    dy = (t1 - t2) * torch.sin(kx)
    alpha = dx + 1j * gamma * torch.cos(kx)
    beta  = dy + 1j * gamma * torch.sin(kx)
    # 2x2 traceless Hamiltonian
    H = torch.zeros(kx.shape + (2, 2), dtype=torch.complex64)
    H[..., 0, 1] = alpha - 1j * beta
    H[..., 1, 0] = alpha + 1j * beta
    return H

@torch.jit.script
def discriminant(kx: torch.Tensor, ky: torch.Tensor, **hop) -> torch.Tensor:
    H = bloch_H(kx, ky, **hop)
    ev = torch.linalg.eigvals(H)
    return (ev[..., 0] - ev[..., 1]).abs()

@torch.jit.script
def ep_winding_proxy(kx: torch.Tensor, ky: torch.Tensor, **hop) -> torch.Tensor:
    # cheap proxy: log-det gradient magnitude around candidate EP
    disc = discriminant(kx, ky, **hop)
    return torch.log(disc + 1e-30)

@torch.jit.script
def skin_weight(evec: torch.Tensor) -> torch.Tensor:
    # evec: (N_sites, N_modes) right eigenvectors
    W = (evec.abs() ** 2).sum(dim=-1)
    return W / (W.sum() + 1e-30)

@torch.jit.script
def gdse_residual(
    W_square: torch.Tensor, W_diamond: torch.Tensor,
    area_pbc: torch.Tensor, area_obc_sq: torch.Tensor, area_obc_dia: torch.Tensor
) -> dict:
    R_area_sq  = (area_obc_sq - area_pbc).abs()
    R_area_dia = (area_obc_dia - area_pbc).abs()
    R_W = (W_diamond.max() - W_square.max()).abs()
    return {
        "R_area_square": R_area_sq,
        "R_area_diamond": R_area_dia,
        "R_W": R_W,
        "scar_floor_feed": R_area_dia + R_W,
        "holographic_DeltaF_trigger": torch.relu(R_area_dia - 1e-4),
        "regulator_on": (R_area_dia > R_area_sq * 10.0),
        "Qin_compatible": W_diamond
    }
```
