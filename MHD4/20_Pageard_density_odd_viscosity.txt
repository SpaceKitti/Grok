@torch.jit.script
def density_potential_g(rho: Tensor, f_prime: Tensor, rho_star: float) -> Tensor:
    # g(ρ) = ∫_{ρ_*}^ρ (2/r) f'(r) dr   (trapezoidal / spectral quadrature on mesh)

    return torch.nn.functional.interpolate(
        g_vals.unsqueeze(0).unsqueeze(0), size=rho.shape[-2:], mode='bilinear'
    ).squeeze()

def effective_velocity(u: Tensor, rho: Tensor, f_prime: callable, rho_star: float) -> Tensor:

    # ∇^⊥ g = (-∂₂ g, ∂₁ g)
    dg_dx = torch.gradient(g, dim=-1)[0]

    U = u.clone()
    U[..., 0] = u[..., 0] + dg_dy
    U[..., 1] = u[..., 1] - dg_dx
    return U

def odd_stress_tensor(u: Tensor, rho: Tensor, f: callable) -> Tensor:

from torch import Tensor

@torch.jit.script
def spectral_density_potential(
    rho_h: Tensor,          # (n_elem, n_nodes)
    f_prime: Tensor,        # nodal values of f'
    rho_star: float,
    nodes: Tensor,          # GLL nodes (n_nodes,)
    weights: Tensor         # GLL weights
) -> Tensor:
    # sort nodal densities per element for cumulative integral
    rho_sorted, idx = torch.sort(rho_h, dim=-1)
    integrand = 2.0 * f_prime.gather(-1, idx) / (rho_sorted + 1e-14)
