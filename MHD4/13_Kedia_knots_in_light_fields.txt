from scipy.special import genlaguerre

def LG(ell, p, rho, phi, z, w0=1.0, k=1.0):
    """Normalized Laguerre-Gaussian mode (paraxial)."""
    zr = k*w0**2/2
    w = w0*np.sqrt(1+(z/zr)**2)
    R = z*(1+(zr/z)**2) if z != 0 else np.inf

    norm = np.sqrt(2*np.math.factorial(p)/(np.pi*np.math.factorial(p+abs(ell)))) / w
    return (norm * (rho*np.sqrt(2)/w)**abs(ell) *
            L(2*rho**2/w**2) *
            np.exp(-rho**2/w**2) *
            np.exp(-1j*k*rho**2/(2*R) if R < np.inf else 0) *
            np.exp(1j*ell*phi) *
