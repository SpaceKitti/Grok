import torch

@torch.jit.script
def integral_scale_B(B: torch.Tensor, dx: float) -> torch.Tensor:
    F = torch.fft.fftn(B, dim=(-4, -3, -2))
    spec = (F.abs() ** 2).sum(-1)
    kx = torch.fft.fftfreq(B.shape[-4], d=dx, device=B.device)
    ky = torch.fft.fftfreq(B.shape[-3], d=dx, device=B.device)
    kz = torch.fft.fftfreq(B.shape[-2], d=dx, device=B.device)
    KX, KY, KZ = torch.meshgrid(kx, ky, kz, indexing="ij")
    k = (KX**2 + KY**2 + KZ**2).sqrt().clamp_min(1e-12)
    num = (spec / k).mean()
    den = spec.mean().clamp_min(1e-30)
    return num / den

@torch.jit.script
def magnetic_helicity_spectral(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    return (A * B).sum(-1).mean()

@torch.jit.script
def saffman_helicity_IH(h: torch.Tensor, dx: float) -> torch.Tensor:
    # I_H ≈ ∫ ⟨h(x) h(x+r)⟩ d³r  via Parseval: (2π)³ |ĥ(k=0-band)|² scaled
    H = torch.fft.fftn(h, dim=(-3, -2, -1))
    # exclude mean (already ~0 for non-helical); integrate low-k power
    power = (H.abs() ** 2)
    return power.mean() * (dx ** 3)

@torch.jit.script
def k_sheet(L: torch.Tensor, VA: torch.Tensor, eta: torch.Tensor) -> torch.Tensor:
    S = (VA * L / eta.clamp_min(1e-30)).clamp_min(1.0)
    return S.sqrt() / L.clamp_min(1e-30)

@torch.jit.script

        "helical": helical, "fast": fast,
