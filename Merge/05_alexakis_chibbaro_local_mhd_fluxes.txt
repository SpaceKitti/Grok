\[

**Four flux channels (Eqs. 16–20)**
\[
\begin{aligned}
\Pi_\ell^{uu}&=-\tau_{\ell,ij}^{uu}\,\partial_i\widetilde{u}_j,\\
\Pi_\ell^{bb}&=+\tau_{\ell,ij}^{bb}\,\partial_i\widetilde{u}_j,\\
\Pi_\ell^{ub}&=-\tau_{\ell,ij}^{ub}\,\partial_i\widetilde{b}_j,\\
\Pi_\ell^{bu}&=+\tau_{\ell,ij}^{bu}\,\partial_i\widetilde{b}_j.
\end{aligned}

import torch

@torch.jit.script
def gaussian_filter_3d(field: torch.Tensor, ell: float, dx: float) -> torch.Tensor:
    F = torch.fft.fftn(field, dim=(-3, -2, -1))
    kx = torch.fft.fftfreq(field.shape[-3], d=dx, device=field.device)
    ky = torch.fft.fftfreq(field.shape[-2], d=dx, device=field.device)
    kz = torch.fft.fftfreq(field.shape[-1], d=dx, device=field.device)
    KX, KY, KZ = torch.meshgrid(kx, ky, kz, indexing="ij")
    G = torch.exp(-0.5 * (ell**2) * (KX**2 + KY**2 + KZ**2))
    return torch.fft.ifftn(F * G, dim=(-3, -2, -1)).real

@torch.jit.script
def sgs_stress(f: torch.Tensor, g: torch.Tensor, ell: float, dx: float) -> torch.Tensor:
    fg = torch.einsum("...i,...j->...ij", f, g)
    return gaussian_filter_3d(fg, ell, dx) - torch.einsum(
        "...i,...j->...ij",
        gaussian_filter_3d(f, ell, dx),
        gaussian_filter_3d(g, ell, dx),
    )

@torch.jit.script
def strain(u: torch.Tensor, dx: float) -> torch.Tensor:
    grad = torch.stack(torch.gradient(u, spacing=dx, dim=(-3, -2, -1)), dim=-1)
    return 0.5 * (grad + grad.transpose(-1, -2))

@torch.jit.script
def vorticity(u: torch.Tensor, dx: float) -> torch.Tensor:
    g = torch.stack(torch.gradient(u, spacing=dx, dim=(-3, -2, -1)), dim=-1)
    return torch.stack((
        g[..., 2, 1] - g[..., 1, 2],
        g[..., 0, 2] - g[..., 2, 0],
        g[..., 1, 0] - g[..., 0, 1],
    ), dim=-1)

@torch.jit.script
def alexakis_chibbaro_fluxes(
    u: torch.Tensor, b: torch.Tensor, ell: float, dx: float
) -> dict:
    ul = gaussian_filter_3d(u, ell, dx)
    bl = gaussian_filter_3d(b, ell, dx)
    tau_uu = sgs_stress(u, u, ell, dx)
    tau_bb = sgs_stress(b, b, ell, dx)
    tau_ub = sgs_stress(u, b, ell, dx)
    tau_bu = sgs_stress(b, u, ell, dx)
    du = torch.stack(torch.gradient(ul, spacing=dx, dim=(-3, -2, -1)), dim=-1)
    db = torch.stack(torch.gradient(bl, spacing=dx, dim=(-3, -2, -1)), dim=-1)
    Pi_uu = -(tau_uu * du).sum(dim=(-1, -2))
    Pi_bb = +(tau_bb * du).sum(dim=(-1, -2))
    Pi_ub = -(tau_ub * db).sum(dim=(-1, -2))
    Pi_bu = +(tau_bu * db).sum(dim=(-1, -2))
    Pi = Pi_uu + Pi_bb + Pi_ub + Pi_bu
    S_b = strain(bl, dx)
    O_b = vorticity(bl, dx)
    bmag = bl.norm(dim=-1)
    return {
        "Pi": Pi, "Pi_uu": Pi_uu, "Pi_bb": Pi_bb, "Pi_ub": Pi_ub, "Pi_bu": Pi_bu,
        "S_b": S_b, "Omega_b": O_b, "bmag": bmag,
        "mag_fraction": (Pi_bb + Pi_bu).abs() / (Pi.abs() + 1e-18),
    }

@torch.jit.script
def ac22_residual(
    u: torch.Tensor, b: torch.Tensor, ell: float, dx: float,
    regime: str = "strong"
) -> dict:
    fl = alexakis_chibbaro_fluxes(u, b, ell, dx)
    Pi = fl["Pi"]
    R_Pi = (Pi - fl["Pi_uu"] - fl["Pi_bb"] - fl["Pi_ub"] - fl["Pi_bu"]).abs().mean()
    Sb2 = (fl["S_b"] ** 2).sum((-2, -1))
    Ob2 = (fl["Omega_b"] ** 2).sum(-1)
    # cheap correlation proxies
    c_Sb = ((Pi - Pi.mean()) * (Sb2 - Sb2.mean())).mean()
    c_Ob = ((Pi - Pi.mean()) * (Ob2 - Ob2.mean())).mean()
    R_grad = (c_Sb.abs() + c_Ob.abs()).relu()  # MHD must be magnetic-dominated
    tail_neg = torch.relu(-Pi).mean()
    R_b = tail_neg * (fl["bmag"] > fl["bmag"].mean()).float().mean()
    scar = R_Pi + (1.0 - fl["mag_fraction"].mean()) + R_b
    return {
        "Pi_mean": Pi.mean(),
        "Pi_skew": ((Pi - Pi.mean()) ** 3).mean() / (Pi.std() ** 3 + 1e-18),
        "mag_fraction": fl["mag_fraction"].mean(),
        "R_Pi": R_Pi,
        "R_grad_magnetic": R_grad,
        "R_|b|_backscatter": R_b,
        "scar_floor_feed": scar,
        "holographic_DeltaF_trigger": torch.relu(scar - 1e-8),
        "regime": regime,
        "Qin_compatible": (u, b),
        "heating_polarity": torch.relu(-Pi),          # forward cascade
        "structure_corr": (Pi.abs() * Sb2).mean(),
    }
