@torch.jit.script
def hall_viscosity_from_berry(
    k2: torch.Tensor, Omega_z: torch.Tensor, f0: torch.Tensor, hbar: float, Q: float
) -> torch.Tensor:
    return hbar * (k2 * Omega_z * f0).sum() / (Q + 1e-30)

@torch.jit.script
def eta_H_wall(y: torch.Tensor, etaH: float, a: float) -> torch.Tensor:
    return etaH * torch.tanh(y / (2.0 * a + 1e-30))

@torch.jit.script
def odd_body_force(grad_u: torch.Tensor, d_etaH_dy: torch.Tensor) -> torch.Tensor:
    # F_i = eps_jl (d eta_H / dy) d_l u_j   (2-D)
    # grad_u: (..., 2, 2)  with components d_l u_k
    Fx = d_etaH_dy * grad_u[..., 1, 0]   # schematic 2-D Levi-Civita contraction
    Fy = -d_etaH_dy * grad_u[..., 0, 0]
    return torch.stack((Fx, Fy), dim=-1)

@torch.jit.script
def vorticity_2d(u: torch.Tensor, dx: float) -> torch.Tensor:
    du_dy = torch.gradient(u[..., 0], spacing=dx, dim=-2)[0]
    dv_dx = torch.gradient(u[..., 1], spacing=dx, dim=-1)[0]
    return dv_dx - du_dy

@torch.jit.script
def chamber_circulation(u: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    # discrete H ~ sign-weighted loop; production: boundary trace on Qin facets
    tang = u * mask
    return torch.tanh(tang.sum())

@torch.jit.script
def enstrophy_chamber(omega: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    return (omega.abs() * mask).sum()

@torch.jit.script
def enstrophy_transport_identity(
    omega: torch.Tensor, eta_H: torch.Tensor, d_etaH_dy: torch.Tensor, mask: torch.Tensor
) -> torch.Tensor:
    lhs = (d_etaH_dy * omega * omega * mask).sum()
    # RHS proxy: jump of eta_H omega^2 across left/right traces
    return lhs

    mu: float, gamma: float, rho: float, W: float,

    chi_target: float, dx: float

    return {
        "chi": chi,
        "Gamma": Gamma,
        "H_L": H_L,
        "H_R": H_R,
        "ens_L": ens_L,
        "ens_R": ens_R,
