**Paper core \(\to\) hive strata.**

Incompressible Elsässer system used throughout the paper and already native to the hive:
\[
\mathbf{w}^\pm=\mathbf{v}\pm\mathbf{b},\qquad
\partial_t\mathbf{w}^\pm+(\mathbf{w}^\mp\cdot\nabla)\mathbf{w}^\pm=-\nabla P_*+\nu_n\nabla^{2n}\mathbf{w}^\pm
\]
(\(n=1\) normal viscosity, \(n>1\) hyperviscosity). Angle-integrated three-dimensional spectrum and one-dimensional spectrum
\[
E_k=\int_{|\mathbf{k}|\in[n,n+1]}F(\mathbf{k})\,d^3k/(4\pi k^2),\qquad
P_k=\int e^{i\mathbf{k}\cdot\mathbf{r}}B(\mathbf{r})\,d\mathbf{r}/\pi,
\]
with the usual relation \(E_k\propto k\,dP_k/dk\). Compensated plots \(E_k k^{3/2}\) and \(P_k k^{3/2}\) are the operational objects.

Normal-viscosity runs: MHD slopes similar to hydro but slightly shallower; dissipation scale \(d_\nu^{\mathrm{MHD}}\approx1.3\,d_\nu^{\mathrm{HD}}\). Hyperviscosity runs: hydro develops a pronounced, well-defined bottleneck pile-up immediately before the dissipation range; MHD spectra remain comparatively flat. The paper interprets the differential bottleneck as direct evidence that the MHD cascade is *less local* than Kolmogorov hydrodynamics, so models that assume a strictly local “as-in-hydro” transfer (GS95, Boldyrev 2006) require revision. Finite-resolution slope measurements at currently available \(512^3\)–\(1024^3\) are therefore systematically contaminated even with ordinary viscosity.

Alignment suite (increments \(\delta\mathbf{w}^\pm\), \(\delta\mathbf{v}\), \(\delta\mathbf{b}\)):
\[
\begin{align*}
\mathrm{AA}&=\langle|\sin\theta|\rangle &&(\theta=\angle(\delta\mathbf{w}^+,\delta\mathbf{w}^-)),\\
\mathrm{AA2}&=\langle|\sin\theta|^2\rangle &&(\theta=\angle(\delta\mathbf{v},\delta\mathbf{b})),\\

**Production kernels (Torch / JAX, Qin-drop-in).**

```python
import torch

@torch.jit.script
def elsasser(v: torch.Tensor, b: torch.Tensor):
    return v + b, v - b

@torch.jit.script
def shell_spectrum(field: torch.Tensor, kx, ky, kz) -> torch.Tensor:
    # field: (3, Nx, Ny, Nz) spectral or real; assume real-space input
    F = torch.fft.fftn(field, dim=(-3, -2, -1))
    power = (F.real**2 + F.imag**2).sum(0)
    k2 = kx[:, None, None]**2 + ky[None, :, None]**2 + kz[None, None, :]**2
    k = torch.sqrt(k2)
    kmax = int(k.max().item()) + 1
    Ek = torch.zeros(kmax, device=field.device)
    for n in range(1, kmax):
        mask = (k >= n) & (k < n + 1)
        Ek[n] = power[mask].sum() / (4.0 * 3.14159265 * float(n)**2 + 1e-30)
    return Ek

@torch.jit.script
def increments(field: torch.Tensor, shift):
    return torch.roll(field, shifts=shift, dims=(-3, -2, -1)) - field

@torch.jit.script
def alignment_suite(v: torch.Tensor, b: torch.Tensor, shift) -> dict:
    dv = increments(v, shift)
    db = increments(b, shift)
    wp, wm = elsasser(dv, db)
    # angles
    cross_w = torch.linalg.norm(torch.cross(wp, wm, dim=0), dim=0)
    den_w = (torch.linalg.norm(wp, dim=0) * torch.linalg.norm(wm, dim=0)).clamp(min=1e-30)
    sin_th_w = (cross_w / den_w).clamp(0.0, 1.0)
    AA = sin_th_w.mean()
    AA2 = (sin_th_w**2).mean()
    PI = (torch.linalg.norm(wp, dim=0) * torch.linalg.norm(wm, dim=0) * sin_th_w).mean() / (
        (torch.linalg.norm(wp, dim=0) * torch.linalg.norm(wm, dim=0)).mean() + 1e-30
    )
    IM = ((torch.linalg.norm(wp, dim=0)**2 - torch.linalg.norm(wm, dim=0)**2).abs()).mean() / (
        (torch.linalg.norm(wp, dim=0)**2 + torch.linalg.norm(wm, dim=0)**2).mean() + 1e-30
    )
    IC = (torch.linalg.norm(wp, dim=0) * torch.linalg.norm(wm, dim=0)).mean() / (
        (torch.linalg.norm(wp, dim=0)**2 - torch.linalg.norm(wm, dim=0)**2).abs().mean() + 1e-30
    )
    FC = (dv * db).sum(0).abs().mean() / (
        (torch.linalg.norm(dv, dim=0)**2).mean().sqrt() * (torch.linalg.norm(db, dim=0)**2).mean().sqrt() + 1e-30
    )
    cross_vb = torch.linalg.norm(torch.cross(dv, db, dim=0), dim=0)
    den_vb = (torch.linalg.norm(dv, dim=0) * torch.linalg.norm(db, dim=0)).clamp(min=1e-30)
    DA = (cross_vb).mean() / (den_vb.mean() + 1e-30)
    return {"AA": AA, "AA2": AA2, "PI": PI, "IM": IM, "IC": IC, "FC": FC, "DA": DA}

@torch.jit.script
def compensated_bottleneck(Ek_hd: torch.Tensor, Ek_mhd: torch.Tensor, k: torch.Tensor, p: float = 1.5):
    comp_hd = Ek_hd * (k ** p)
    comp_mhd = Ek_mhd * (k ** p)
    # bottleneck window: last decade before dissipation roll-off
    n = k.numel()
    sl = slice(n // 2, int(0.85 * n))
    R_BN = (comp_hd[sl] - comp_mhd[sl]).abs().mean()
    return {"comp_hd": comp_hd, "comp_mhd": comp_mhd, "R_BN": R_BN}

@torch.jit.script
def beresnyak_lazarian_residual(
    v: torch.Tensor, b: torch.Tensor,
    Ek_hd: torch.Tensor, Ek_mhd: torch.Tensor, k: torch.Tensor,
    shifts: torch.Tensor, dx: float
) -> dict:
    acc = {key: torch.zeros(()) for key in ["AA", "AA2", "PI", "IM", "IC", "FC", "DA"]}
    nsh = shifts.shape[0]
    i = 0
    while i < nsh:
        sh = (int(shifts[i, 0].item()), int(shifts[i, 1].item()), int(shifts[i, 2].item()))
        al = alignment_suite(v, b, sh)
        for key in acc:
            acc[key] = acc[key] + al[key]
        i += 1
    n = float(nsh)
    bn = compensated_bottleneck(Ek_hd, Ek_mhd, k)
    R_DA = (acc["DA"] / n - 0.0).abs()  # optional Boldyrev prior; do not enforce
    return {
        "AA": acc["AA"] / n,
        "DA": acc["DA"] / n,
        "PI": acc["PI"] / n,
        "R_BN": bn["R_BN"],
        "R_loc": bn["R_BN"] + R_DA,
        "scar_floor_feed": bn["R_BN"],
        "holographic_DeltaF_trigger": torch.relu(bn["R_BN"] - 1e-4),
        "Qin_compatible": v,
    }
```
