### Production kernels (Torch / JAX, Qin-drop-in)

```python
import torch

@torch.jit.script
def gaussian_filter_3d(field: torch.Tensor, ell: float, dx: float) -> torch.Tensor:
    # spectral Gaussian of width ell (periodic box)
    F = torch.fft.fftn(field, dim=(-3, -2, -1))
    kx = torch.fft.fftfreq(field.shape[-3], d=dx, device=field.device)
    ky = torch.fft.fftfreq(field.shape[-2], d=dx, device=field.device)
    kz = torch.fft.fftfreq(field.shape[-1], d=dx, device=field.device)
    KX, KY, KZ = torch.meshgrid(kx, ky, kz, indexing="ij")
    k2 = KX**2 + KY**2 + KZ**2
    G = torch.exp(-0.5 * (ell**2) * k2)
    return torch.fft.ifftn(F * G, dim=(-3, -2, -1)).real

@torch.jit.script
def sgs_stress(f: torch.Tensor, g: torch.Tensor, ell: float, dx: float) -> torch.Tensor:
    # tau(f,g) = filter(f*g) - filter(f)*filter(g)   (last two indices i,j)
    fg = torch.einsum("...i,...j->...ij", f, g)
    return gaussian_filter_3d(fg, ell, dx) - torch.einsum(
        "...i,...j->...ij", gaussian_filter_3d(f, ell, dx), gaussian_filter_3d(g, ell, dx)
    )

@torch.jit.script
def strain(u: torch.Tensor, dx: float) -> torch.Tensor:
    # S_ij = (du_i/dx_j + du_j/dx_i)/2
    grad = torch.stack(torch.gradient(u, spacing=dx, dim=(-3, -2, -1)), dim=-1)
    return 0.5 * (grad + grad.transpose(-1, -2))

@torch.jit.script
def conversion_split(u: torch.Tensor, b: torch.Tensor, B0: float, ell: float, dx: float) -> dict:
    ul = gaussian_filter_3d(u, ell, dx)
    bl = gaussian_filter_3d(b, ell, dx)
    S = strain(ul, dx)
    W_iso = (bl.unsqueeze(-1) * S * bl.unsqueeze(-2)).sum(dim=(-1, -2)).mean()
    W_anis = B0 * (bl * torch.gradient(ul, spacing=dx, dim=-3)[0]).sum(-1).mean()
    return {"W_iso": W_iso, "W_anis": W_anis, "W": W_iso + W_anis}

@torch.jit.script
def fluxes(u: torch.Tensor, b: torch.Tensor, ell: float, dx: float) -> dict:
    ul = gaussian_filter_3d(u, ell, dx)
    bl = gaussian_filter_3d(b, ell, dx)
    S = strain(ul, dx)
    tau_uu = sgs_stress(u, u, ell, dx)
    tau_bb = sgs_stress(b, b, ell, dx)
    tau_bu = sgs_stress(b, u, ell, dx)
    tau_ub = sgs_stress(u, b, ell, dx)
    Pi_I = -(tau_uu * S).sum(dim=(-1, -2)).mean()
    Pi_M = -(tau_bb * S).sum(dim=(-1, -2)).mean()
    db = torch.stack(torch.gradient(bl, spacing=dx, dim=(-3, -2, -1)), dim=-1)
    Pi_A = -(tau_bu * db).sum(dim=(-1, -2)).mean()
    Pi_D = -(tau_ub * db).sum(dim=(-1, -2)).mean()
    return {"Pi_I": Pi_I, "Pi_M": Pi_M, "Pi_A": Pi_A, "Pi_D": Pi_D}

@torch.jit.script
def col26_residual(
    u: torch.Tensor, b: torch.Tensor, B0: float, ell: float, dx: float,
    Pi_thin_num: float, Pi_tot: float
) -> dict:
    W = conversion_split(u, b, B0, ell, dx)
    fl = fluxes(u, b, ell, dx)
    R_W = (W["W"] - W["W_iso"] - W["W_anis"]).abs()
    R_anis_large = torch.relu(-W["W_anis"])   # should be >0 at large ell
    R_thin = (Pi_thin_num - fl["Pi_A"]).abs() + (Pi_thin_num / (Pi_tot + 1e-30) - 1.0).abs()
    scar = R_W + R_anis_large + R_thin
    return {
        "W_iso": W["W_iso"],
        "W_anis": W["W_anis"],
        "Pi_I": fl["Pi_I"],
        "Pi_M": fl["Pi_M"],
        "Pi_A": fl["Pi_A"],
        "Pi_D": fl["Pi_D"],
        "R_W": R_W,
        "R_anis": R_anis_large,
        "R_thin": R_thin,
        "scar_floor_feed": scar,
        "holographic_DeltaF_trigger": torch.relu(R_thin - 1e-8),
        "regime": "thinning-dominated" if R_thin < 0.15 else "mixed",
        "Qin_compatible": (u, b),
    }
```
