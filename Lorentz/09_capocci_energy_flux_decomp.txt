import torch

@torch.jit.script
def gaussian_filter(field: torch.Tensor, ell: float, kx, ky, kz) -> torch.Tensor:
    k2 = kx[:, None, None]**2 + ky[None, :, None]**2 + kz[None, None, :]**2
    G = torch.exp(-0.5 * ell**2 * k2)
    return torch.fft.ifftn(G * torch.fft.fftn(field, dim=(-3,-2,-1)), dim=(-3,-2,-1)).real

@torch.jit.script
def strain_rotation(gradU: torch.Tensor) -> tuple:
    S = 0.5 * (gradU + gradU.transpose(-1, -2))
    O = 0.5 * (gradU - gradU.transpose(-1, -2))
    return S, O

@torch.jit.script
def mag_strain_current(gradB: torch.Tensor) -> tuple:
    Sigma = 0.5 * (gradB + gradB.transpose(-1, -2))
    # J_k = eps_kij d_i b_j  (vector)
    J = torch.stack([
        gradB[..., 2, 1] - gradB[..., 1, 2],
        gradB[..., 0, 2] - gradB[..., 2, 0],
        gradB[..., 1, 0] - gradB[..., 0, 1],
    ], dim=-1)
    return Sigma, J

@torch.jit.script
def sgs_stress_gaussian(f: torch.Tensor, g: torch.Tensor, ell: float) -> torch.Tensor:
    # leading single-scale piece; multi-scale integral left as residual monitor
    fg = torch.einsum("...i,...j->...ij", f, g)
    fbar = gaussian_filter(f, ell, *torch.meshgrid(
        torch.fft.fftfreq(f.shape[-3]), torch.fft.fftfreq(f.shape[-2]), torch.fft.fftfreq(f.shape[-1]), indexing="ij"
    )[:3]) if False else f  # placeholder; production uses precomputed k-grid
    return ell**2 * torch.einsum("...ik,...jk->...ij", torch.gradient(g)[0], torch.gradient(f)[0])

@torch.jit.script
def maxwell_subfluxes(S: torch.Tensor, Sigma: torch.Tensor, J: torch.Tensor, ell: float) -> dict:
    # traces of the three Maxwell channels (single-scale pieces)
    SSigmaSigma = ell**2 * torch.einsum("...ij,...jk,...ki->...", S, Sigma, Sigma)
    SJJ = ell**2 * torch.einsum("...ij,...j,...i->...", S, J, J)
    SJSigma = ell**2 * torch.einsum("...ij,...j,...ik->...", S, J, Sigma)
    return {"SSigmaSigma": SSigmaSigma.mean(), "SJJ": SJJ.mean(), "SJSigma": SJSigma.mean()}

@torch.jit.script
def resolved_conversion(u_ell: torch.Tensor, J_ell: torch.Tensor, B_ell: torch.Tensor) -> torch.Tensor:
    return (u_ell * torch.cross(J_ell, B_ell, dim=-1)).sum()

@torch.jit.script

    ell: float, W_L_global: torch.Tensor

    R_WL = (W_ell + torch.einsum("...i,...ij,...j->...", B, S, B).mean()).abs()

    return {
        "W_ell": W_ell,
        "SJSigma": sub["SJSigma"],
        "SJJ": sub["SJJ"],
        "SSigmaSigma": sub["SSigmaSigma"],
        "R_WL": R_WL,
        "R_SJSigma": R_SJSigma,
