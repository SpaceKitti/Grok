import torch

@torch.jit.script
def swirl_norm(u_th: torch.Tensor, r: torch.Tensor) -> torch.Tensor:
    return u_th / (r + 1e-30)

@torch.jit.script
def reconstruct_poloidal(psi1: torch.Tensor, r: torch.Tensor, dz: torch.Tensor, dr: torch.Tensor):
    ur = -r * dz
    uz = 2.0 * psi1 + r * dr
    return ur, uz

@torch.jit.script
def cks_integrand(omega: torch.Tensor, j: torch.Tensor) -> torch.Tensor:
    return omega.abs().max() + j.abs().max()

@torch.jit.script
def wall_spectrum(field_r: torch.Tensor, field_th: torch.Tensor, field_z: torch.Tensor) -> torch.Tensor:
    # r=1 slice, Fourier in z
    fr = torch.fft.rfft(field_r[-1], dim=-1)
    fth = torch.fft.rfft(field_th[-1], dim=-1)
    fz = torch.fft.rfft(field_z[-1], dim=-1)
    S = (fr.abs()**2 + fth.abs()**2 + fz.abs()**2)
    g = torch.ones_like(S)
    g[1:] = 2.0
    return 0.5 * g * S / field_z.shape[-1]

@torch.jit.script
def analyticity_strip(S: torch.Tensor, k: torch.Tensor, kmin: int = 8) -> torch.Tensor:
    # log S ~ const - k * delta  (linear fit on tail)
    logS = torch.log(S[kmin:] + 1e-30)
    kk = k[kmin:].to(S.dtype)
    # least-squares slope
    kmean = kk.mean()
    smean = logS.mean()
    cov = ((kk - kmean) * (logS - smean)).sum()
    var = ((kk - kmean)**2).sum()
    slope = cov / (var + 1e-30)
    delta = -slope
    return torch.relu(delta)

@torch.jit.script
def pressure_wall_residual(u_th: torch.Tensor, b_th: torch.Tensor, dp_dr: torch.Tensor) -> torch.Tensor:
    return (dp_dr - u_th**2 + b_th**2).abs().mean()

@torch.jit.script
def kp25_residual(
    u_th: torch.Tensor, b_th: torch.Tensor, omega: torch.Tensor, j: torch.Tensor,
    S_T: torch.Tensor, k: torch.Tensor, C: float, C_target: float,
    delta: torch.Tensor, t: float, tC: float, gamma: float,
    dp_dr: torch.Tensor
) -> dict:
    R_C = abs(C - C_target)
    R_CKS = cks_integrand(omega, j)
    R_delta = analyticity_strip(S_T, k)
    power_law = (tC - t).clamp(min=1e-12)**gamma
    R_fit = (R_delta - power_law).abs()
    R_p = pressure_wall_residual(u_th[-1], b_th[-1], dp_dr)
    R_stat = (u_th.abs() + b_th.abs()).mean() * float(abs(C - 1.0) < 1e-8)
    scar = R_C + R_CKS + R_delta + R_fit + R_p
    return {
        "C": C,
        "R_C": R_C,
        "R_CKS": R_CKS,
        "delta": R_delta,
        "R_fit": R_fit,
        "R_p": R_p,
        "R_stat": R_stat,
        "scar_floor_feed": scar,
        "holographic_DeltaF_trigger": torch.relu(scar - 1e-8),
        "morphology": "square" if C < 1.0 else ("cusp" if C > 1.0 else "stationary"),
        "Qin_compatible": (u_th, b_th),
    }
