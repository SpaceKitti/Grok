### Production kernels (Torch / JAX, Qin-drop-in)

```python
import torch

@torch.jit.script
def elsasser(v: torch.Tensor, b: torch.Tensor) -> tuple:
    return v + b, v - b

@torch.jit.script
def increments(field: torch.Tensor, shift: tuple) -> torch.Tensor:
    # periodic roll; shift = (sx, sy, sz) in cells
    return torch.roll(field, shifts=shift, dims=(0, 1, 2)) - field

@torch.jit.script
def third_order_yaglom(
    zp: torch.Tensor, zm: torch.Tensor, shift: tuple, dx: float
) -> tuple:
    dzp = increments(zp, shift)
    dzm = increments(zm, shift)
    rvec = torch.tensor([float(shift[0]), float(shift[1]), float(shift[2])]) * dx
    r = torch.linalg.norm(rvec).clamp(min=1e-30)
    rhat = rvec / r
    flux_p = (dzm * (dzp * dzp).sum(-1, keepdim=True)).sum(-1)  # δz- |δz+|²  (vector)
    flux_m = (dzp * (dzm * dzm).sum(-1, keepdim=True)).sum(-1)
    long_p = (dzm * rhat).sum(-1) * (dzp * dzp).sum(-1)
    long_m = (dzp * rhat).sum(-1) * (dzm * dzm).sum(-1)
    return flux_p.mean(), flux_m.mean(), long_p.mean(), long_m.mean(), r

@torch.jit.script
def mixed_third_order(v: torch.Tensor, b: torch.Tensor, shift: tuple, dx: float) -> tuple:
    dv = increments(v, shift)
    db = increments(b, shift)
    rvec = torch.tensor([float(shift[0]), float(shift[1]), float(shift[2])]) * dx
    r = torch.linalg.norm(rvec).clamp(min=1e-30)
    rhat = rvec / r
    dvL = (dv * rhat).sum(-1)
    dbL = (db * rhat).sum(-1)
    mix = dvL * ((dv * dv).sum(-1) + (db * db).sum(-1)) - 2.0 * dbL * (dv * db).sum(-1)
    return mix.mean(), r

@torch.jit.script
def dissipation_split(
    omega: torch.Tensor, J: torch.Tensor, gradJ: torch.Tensor,
    nu: float, eta: float, eta_H: float
) -> tuple:
    eps_nu = nu * (omega * omega).mean()
    eps_eta = eta * (J * J).mean()
    eps_H = eta_H * (gradJ * gradJ).mean()
    eps = eps_nu + eps_eta + eps_H
    return eps_nu, eps_eta, eps_H, eps

@torch.jit.script
def alignment_angle(dv: torch.Tensor, db: torch.Tensor) -> torch.Tensor:
    cross = torch.linalg.norm(torch.cross(dv, db, dim=-1), dim=-1)
    denom = (torch.linalg.norm(dv, dim=-1) * torch.linalg.norm(db, dim=-1)).clamp(min=1e-30)
    return torch.asin((cross / denom).clamp(0.0, 1.0))

@torch.jit.script
def politano_pouquet_residual(
    v: torch.Tensor, b: torch.Tensor, omega: torch.Tensor, J: torch.Tensor,
    gradJ: torch.Tensor, shifts: torch.Tensor, dx: float,
    nu: float, eta: float, eta_H: float
) -> dict:
    zp, zm = elsasser(v, b)
    eps_nu, eps_eta, eps_H, eps = dissipation_split(omega, J, gradJ, nu, eta, eta_H)
    # accumulate over a shell of isotropic shifts
    acc_long_p = torch.zeros(())
    acc_long_m = torch.zeros(())
    acc_mix = torch.zeros(())
    acc_r = torch.zeros(())
    nsh = shifts.shape[0]
    i = 0
    while i < nsh:
        sh = (int(shifts[i, 0].item()), int(shifts[i, 1].item()), int(shifts[i, 2].item()))
        _, _, lp, lm, r = third_order_yaglom(zp, zm, sh, dx)
        mix, _ = mixed_third_order(v, b, sh, dx)
        acc_long_p = acc_long_p + lp
        acc_long_m = acc_long_m + lm
        acc_mix = acc_mix + mix
        acc_r = acc_r + r
        i += 1
    n = float(nsh)
    long_p = acc_long_p / n
    long_m = acc_long_m / n
    mix = acc_mix / n
    rbar = acc_r / n
    # split ε± ≈ ε/2 in the near-equipartition HIT regime; expose both
    eps_p = 0.5 * eps
    eps_m = 0.5 * eps
    R43_p = (long_p + (4.0 / 3.0) * eps_p * rbar).abs()
    R43_m = (long_m + (4.0 / 3.0) * eps_m * rbar).abs()
    Rmix = (mix + (4.0 / 3.0) * eps * rbar).abs()
    ell = mix.abs() / (eps + 1e-30)
    R_ell = (ell - (4.0 / 3.0) * rbar).abs()
    return {
        "eps": eps,
        "eps_nu": eps_nu,
        "eps_eta": eps_eta,
        "eps_H": eps_H,
        "r": rbar,
        "S3_long_plus": long_p,
        "S3_long_minus": long_m,
        "S3_mix": mix,
        "ell_eps": ell,
        "R_4_3_plus": R43_p,
        "R_4_3_minus": R43_m,
        "R_mix": Rmix,
        "R_ell": R_ell,
        "scar_floor_feed": R43_p + R43_m + Rmix + R_ell,
        "holographic_DeltaF_trigger": torch.relu(Rmix - 1e-8),
        "Qin_compatible": zp,
    }
```
