@torch.jit.script
def upper_convected(T: torch.Tensor, U: torch.Tensor, gradU: torch.Tensor) -> torch.Tensor:
    # T: (...,3,3), gradU: (...,3,3)
    conv = torch.einsum("...k,...kij->...ij", U, torch.gradient(T)[0])  # schematic advective
    return conv - torch.einsum("...ki,...kj->...ij", gradU, T) - torch.einsum("...ik,...kj->...ij", T, gradU)

def lorentz_work(u: torch.Tensor, J: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
