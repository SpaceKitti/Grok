import torch

@torch.jit.script
def littlewood_paley_project(field: torch.Tensor, j: int, dx: float) -> torch.Tensor:
    # FFT-based dyadic projector (annulus |k| ~ 2^j)
    F = torch.fft.fftn(field, dim=(-3, -2, -1))
    kx = torch.fft.fftfreq(field.shape[-3], d=dx, device=field.device)
    ky = torch.fft.fftfreq(field.shape[-2], d=dx, device=field.device)
    kz = torch.fft.fftfreq(field.shape[-1], d=dx, device=field.device)
    KX, KY, KZ = torch.meshgrid(kx, ky, kz, indexing="ij")
    kabs = torch.sqrt(KX**2 + KY**2 + KZ**2 + 1e-30)
    mask = ((kabs > 0.75 * 2**j) & (kabs < 2.0 * 2**j)).to(F.dtype)
    return torch.fft.ifftn(F * mask, dim=(-3, -2, -1)).real

@torch.jit.script
def vorticity(u: torch.Tensor, dx: float) -> torch.Tensor:
    # discrete curl on collocated or staggered Qin-compatible grid
    duz_dy = torch.gradient(u[..., 2], spacing=dx, dim=-2)[0]
    duy_dz = torch.gradient(u[..., 1], spacing=dx, dim=-1)[0]
    dux_dz = torch.gradient(u[..., 0], spacing=dx, dim=-1)[0]
    duz_dx = torch.gradient(u[..., 2], spacing=dx, dim=-3)[0]
    duy_dx = torch.gradient(u[..., 1], spacing=dx, dim=-3)[0]
    dux_dy = torch.gradient(u[..., 0], spacing=dx, dim=-2)[0]
    wx = duz_dy - duy_dz
    wy = dux_dz - duz_dx
    wz = duy_dx - dux_dy
    return torch.stack((wx, wy, wz), dim=-1)

@torch.jit.script
def current_density(b: torch.Tensor, dx: float) -> torch.Tensor:
    return vorticity(b, dx)   # j = curl b (units mu0=1)

@torch.jit.script
def lp_vorticity_Linf(omega: torch.Tensor, j_max: int, dx: float) -> torch.Tensor:
    vals = []
    for j in range(-j_max, j_max + 1):
        dj_w = littlewood_paley_project(omega, j, dx)
        vals.append(dj_w.abs().amax())
    return torch.stack(vals).amax()

@torch.jit.script
def cmz07_residual(
    u: torch.Tensor, b: torch.Tensor, dt: float, t_acc: torch.Tensor,
    I_BKM_acc: torch.Tensor, dx: float, j_max: int, M_crit: float
) -> dict:
    omega = vorticity(u, dx)
    j = current_density(b, dx)
    I_loc = lp_vorticity_Linf(omega, j_max, dx)
    I_BKM = I_BKM_acc + I_loc * dt
    R_BKM = torch.relu(I_BKM - M_crit)
    jmax = j.abs().amax()
    wmax = omega.abs().amax()
    R_spike = torch.relu(jmax - wmax) * (R_BKM < 1e-12).to(jmax.dtype)
    scar = R_BKM + R_spike
    return {
        "I_BKM": I_BKM,
        "I_loc": I_loc,
        "jmax": jmax,
        "wmax": wmax,
        "R_BKM": R_BKM,
        "R_j_spike": R_spike,
        "scar_floor_feed": scar,
        "holographic_DeltaF_trigger": torch.relu(R_spike - 1e-8),
        "regime": "dissipative-sheet" if (R_BKM < 1e-12 and R_spike > 0) else (
            "BKM-blowup" if R_BKM > 0 else "smooth"
        ),
        "Qin_compatible": (u, b),
    }
