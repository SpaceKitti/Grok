import torch

@torch.jit.script
def current_spectrum(A: torch.Tensor, kx, ky) -> torch.Tensor:
    # A: flux, 2-D; J ~ -lap A
    Ahat = torch.fft.fftn(A)
    k2 = kx[:, None]**2 + ky[None, :]**2
    Jhat = k2 * Ahat
    power = (Jhat.real**2 + Jhat.imag**2)
    k = torch.sqrt(k2)
    kmax = int(k.max().item()) + 1
    EJ = torch.zeros(kmax, device=A.device)
    for n in range(1, kmax):
        mask = (k >= n) & (k < n + 1)
        EJ[n] = power[mask].sum()
    return EJ

@torch.jit.script
def seed_flux(A: torch.Tensor, eps: float, ksup: float, kx, ky) -> torch.Tensor:
    noise = torch.randn_like(A)
    nhat = torch.fft.fftn(noise)
    k = torch.sqrt(kx[:, None]**2 + ky[None, :]**2)
    nhat = nhat * (k <= ksup).to(A.dtype)
    nreal = torch.fft.ifftn(nhat).real
    nreal = nreal / (nreal.abs().max() + 1e-30)
    return A + eps * nreal

@torch.jit.script
def island_amplitude(A: torch.Tensor, A0: torch.Tensor) -> torch.Tensor:
    return (A - A0).abs().max()

@torch.jit.script
def baty_trigger_residual(
    A: torch.Tensor, A0: torch.Tensor, omega: torch.Tensor,
    t: float, t_peak: float, eps: float, ksup: float,
    kx, ky, eta: float, Lcs: float, VA: float,
    gamma_num: float, Np_num: float
) -> dict:

    kmax = EJ.numel() - 1
    R_EJ = torch.relu(10.0 * EJ[kmax] - EJ.max())
    S = Lcs * VA / (eta + 1e-30)
    R_t = torch.relu(torch.tensor(t_peak - t))
    R_eps = torch.relu(torch.tensor(1e-5 - eps))
    R_k = torch.relu(torch.tensor(64.0 - ksup))  # unstable band proxy
    dA = island_amplitude(A, A0)
    # Comisso-like proxy: gamma ~ S^{1/4} * (aspect evolution); store measured

    R_Np = (Np_num - 11.0).abs()
    R_phys = R_t + R_eps + R_k + R_gamma + R_Np
    return {
        "S": S,
        "EJ_max_ratio": EJ.max() / (EJ[kmax] + 1e-30),
        "R_EJ": R_EJ,
        "R_phys": R_phys,
        "deltaA": dA,
        "scar_floor_feed": R_phys + R_EJ,

        "Qin_compatible": A,
